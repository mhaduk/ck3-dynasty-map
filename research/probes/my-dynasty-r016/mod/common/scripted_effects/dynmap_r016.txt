dynmap_r016_paint_dark_effect = {
	if = {
		limit = { exists = scope:dynmap_r016_audit_armed }
		dynmap_r016_count_role_effect = { ROLE = primary }
		debug_log = "DYNMAP_R016 AUDIT V4 RESULT PRIMARY RGB 25 105 50 WRITES_SUPPRESSED"
		debug_log_scopes = no
	}
	else = {
		if = { limit = { exists = scope:dynmap_r016_diagnostic_armed } set_local_variable = { name = dynmap_r016_d_county_writes value = 0 } }
		every_in_de_jure_hierarchy = {
			continue = { always = yes }
			limit = { tier = tier_barony }
			set_title_color = { 25 105 50 }
			if = {
				limit = { exists = scope:dynmap_r016_diagnostic_armed }
				change_local_variable = { name = dynmap_r016_d_county_writes add = 1 }
				change_local_variable = { name = dynmap_r016_d_attempted_writes add = 1 }
			}
		}
		if = { limit = { exists = scope:dynmap_r016_diagnostic_armed } dynmap_r016_diagnostic_county_primary_effect = yes }
	}
}

# County holder first; missing holder Dynasty never suppresses senior checks.
dynmap_r016_realms_county_effect = {
	if = {
		limit = { exists = holder }
		if = {
			limit = {
				holder = {
					OR = {
						AND = {
							top_liege = this
							trigger_if = {
								limit = {
									exists = dynasty
								}
								dynasty = scope:dynmap_r016_player_dynasty
							}
							trigger_else = {
								always = no
							}
						}
						any_liege_or_above = {
							top_liege = this
							trigger_if = {
								limit = {
									exists = dynasty
								}
								dynasty = scope:dynmap_r016_player_dynasty
							}
							trigger_else = {
								always = no
							}
						}
					}
				}
			}
			dynmap_r016_paint_dark_effect = yes
		}
		else_if = {
			limit = {
				holder = {
					OR = {
						trigger_if = {
							limit = {
								exists = dynasty
							}
							dynasty = scope:dynmap_r016_player_dynasty
						}
						trigger_else = {
							always = no
						}
						any_liege_or_above = {
							trigger_if = {
								limit = {
									exists = dynasty
								}
								dynasty = scope:dynmap_r016_player_dynasty
							}
							trigger_else = {
								always = no
							}
						}
					}
				}
			}
			dynmap_r016_paint_light_effect = yes
		}
		else = {

# County scope, called by the actual selected writer. No inferred pixel/storage data.
dynmap_r016_count_role_effect = {
	change_local_variable = { name = dynmap_r016_d_total add = 1 }
	change_local_variable = { name = dynmap_r016_d_$ROLE$ add = 1 }
	if = {
		limit = { exists = holder }
		change_local_variable = { name = dynmap_r016_d_held add = 1 }
		holder = {
			if = { limit = { NOT = { exists = dynasty } } change_local_variable = { name = dynmap_r016_d_holder_no_dynasty add = 1 } }
			if = { limit = { trigger_if = { limit = { exists = dynasty } dynasty = scope:dynmap_r016_player_dynasty } trigger_else = { always = no } } change_local_variable = { name = dynmap_r016_d_holder_match add = 1 } }
			if = { limit = { any_liege_or_above = { trigger_if = { limit = { exists = dynasty } dynasty = scope:dynmap_r016_player_dynasty } trigger_else = { always = no } } } change_local_variable = { name = dynmap_r016_d_senior_match add = 1 } }
			if = { limit = { trigger_if = { limit = { exists = top_liege } top_liege = { trigger_if = { limit = { exists = dynasty } dynasty = scope:dynmap_r016_player_dynasty } trigger_else = { always = no } } } trigger_else = { always = no } } change_local_variable = { name = dynmap_r016_d_top_match add = 1 } }
			if = { limit = { OR = { this = scope:dynmap_r016_player_root any_liege_or_above = { this = scope:dynmap_r016_player_root } } } change_local_variable = { name = dynmap_r016_d_subtree add = 1 } }
		}
	}
	else = { change_local_variable = { name = dynmap_r016_d_unheld add = 1 } }
}

		else_if = {
			limit = {
				local_var:dynmap_r016_d_details < 20
				OR = {
					local_var:dynmap_r016_d_county_writes = 0
					NOT = { exists = holder }
					trigger_if = { limit = { exists = holder } holder = { OR = { NOT = { exists = dynasty } NOT = { exists = top_liege } } } } trigger_else = { always = no }
				}
			}
			change_local_variable = { name = dynmap_r016_d_details add = 1 }

dynmap_r016_audit_v3_actor_effect = {
	debug_log = "DYNMAP_R016 AUDIT V4 SUBJECT"
	debug_log_scopes = no
	if = {
		limit = { exists = dynasty }
		dynasty = { debug_log = "DYNMAP_R016 AUDIT V4 DYNASTY" debug_log_scopes = no }
	}
	else = { debug_log = "DYNMAP_R016 AUDIT V4 DYNASTY_NULL" }
	if = {
		limit = { exists = liege }
		liege = { debug_log = "DYNMAP_R016 AUDIT V4 LIEGE" debug_log_scopes = no }
		debug_log = "DYNMAP_R016 AUDIT V4 NO_LIEGE NO"
		if = { limit = { liege = this } debug_log = "DYNMAP_R016 AUDIT V4 LIEGE_SELF YES" }
		else = { debug_log = "DYNMAP_R016 AUDIT V4 LIEGE_SELF NO" }
	}
	else = {
		debug_log = "DYNMAP_R016 AUDIT V4 LIEGE_NULL"
		debug_log = "DYNMAP_R016 AUDIT V4 NO_LIEGE YES"
		debug_log = "DYNMAP_R016 AUDIT V4 LIEGE_SELF NO"
	}
	if = {
		limit = { exists = top_liege }
		top_liege = { debug_log = "DYNMAP_R016 AUDIT V4 TOP" debug_log_scopes = no }
		if = { limit = { top_liege = this } debug_log = "DYNMAP_R016 AUDIT V4 TOP_SELF YES" }
		else = { debug_log = "DYNMAP_R016 AUDIT V4 TOP_SELF NO" }
	}
	else = {
		debug_log = "DYNMAP_R016 AUDIT V4 TOP_NULL"
		debug_log = "DYNMAP_R016 AUDIT V4 TOP_SELF NO"
	}
	if = { limit = { is_independent_ruler = yes } debug_log = "DYNMAP_R016 AUDIT V4 INDEPENDENT YES" }
	else = { debug_log = "DYNMAP_R016 AUDIT V4 INDEPENDENT NO" }
}

# Character scope. Local list holds only references, and is cleared on entry/exit.
# The subject itself and every ordinary senior are visited while building roster.
dynmap_r016_audit_v3_intern_actor_effect = {
	if = {
		limit = { NOT = { is_target_in_local_variable_list = { name = dynmap_r016_audit_seen target = this } } }
		add_to_local_variable_list = { name = dynmap_r016_audit_seen target = this }
		change_local_variable = { name = dynmap_r016_d_actors add = 1 }
		debug_log = "DYNMAP_R016 AUDIT V4 ACTOR_BEGIN DEF"
		dynmap_r016_audit_v3_actor_effect = yes
		debug_log = "DYNMAP_R016 AUDIT V4 ACTOR_END"
	}
}

dynmap_r016_audit_v3_roster_county_effect = {
	if = {
		limit = { exists = holder }
		holder = {
			dynmap_r016_audit_v3_intern_actor_effect = yes
			every_liege_or_above = { dynmap_r016_audit_v3_intern_actor_effect = yes }
		}
	}
	change_local_variable = { name = dynmap_r016_d_roster add = 1 }
	debug_log = "DYNMAP_R016 AUDIT V4 COUNTY_BEGIN"
	debug_log_scopes = no
	if = {
		limit = { exists = holder }
		holder = { debug_log = "DYNMAP_R016 AUDIT V4 HOLDER" debug_log_scopes = no }
	}
	else = { debug_log = "DYNMAP_R016 AUDIT V4 HOLDER_NULL" }
	debug_log = "DYNMAP_R016 AUDIT V4 COUNTY_END"
}

# V3 selected-view audit. Shared actor/roster setup; one classifier only.
# No paint, restore, sample, SetMapMode or other-view classifier.

dynmap_r016_audit_v3_roster_effect = {
	every_player = {
		limit = { this = root is_alive = yes exists = dynasty }
		save_scope_as = dynmap_r016_audit_armed
	}
	if = {
		limit = { exists = scope:dynmap_r016_audit_armed }
		scope:dynmap_r016_audit_armed = {
			set_local_variable = { name = dynmap_r016_d_actors value = 1 }
			set_local_variable = { name = dynmap_r016_d_roster value = 0 }
			add_to_local_variable_list = { name = dynmap_r016_audit_seen target = this }
			debug_log = "DYNMAP_R016 AUDIT V4 ACTOR_BEGIN PLAYER"
			dynmap_r016_audit_v3_actor_effect = yes
			debug_log = "DYNMAP_R016 AUDIT V4 ACTOR_END"
			# Player's ordinary ancestry is needed even outside any county holder chain.
			every_liege_or_above = { dynmap_r016_audit_v3_intern_actor_effect = yes }
			debug_log = "DYNMAP_R016 AUDIT V4 ROSTER_BEGIN"
			every_county = { dynmap_r016_audit_v3_roster_county_effect = yes }
			save_scope_value_as = { name = dynmap_r016_out_roster value = local_var:dynmap_r016_d_roster }
			save_scope_value_as = { name = dynmap_r016_out_actors value = local_var:dynmap_r016_d_actors }
			debug_log = "DYNMAP_R016 AUDIT V4 COUNTERS_BEGIN PHASE ROSTER VIEW SELECTED"
			debug_log_scopes = yes
			debug_log = "DYNMAP_R016 AUDIT V4 COUNTERS_END PHASE ROSTER VIEW SELECTED"
			clear_saved_scope = dynmap_r016_out_roster
			clear_saved_scope = dynmap_r016_out_actors
			debug_log = "DYNMAP_R016 AUDIT V4 ROSTER_END"
		}
	}
}
